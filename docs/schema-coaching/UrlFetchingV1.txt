Data Schema (Relational / TypeScript Data Model)The site's content and booking architecture map cleanly to the following core entities:TypeScript// 1. Treatment Category Enum
type TreatmentCategory = 'Laser' | 'Skin' | 'Anti-Ageing' | 'Body' | 'Wellness';

type ConcernFilterTag = 
  | 'Unwanted hair' 
  | 'Ageing and texture' 
  | 'Pigment and veins' 
  | 'Acne' 
  | 'Tattoos';

// 2. Treatment Entity
interface Treatment {
  id: string;                        // e.g., "laser-hair-removal"
  slug: string;                      // URL slug under /treatments/
  name: string;                      // e.g., "Laser Hair Removal"
  category: TreatmentCategory;       // Primary navigation category
  filterTag?: ConcernFilterTag;      // Homepage tab filter
  isCoreFeatured: boolean;           // True for the 8 homepage priced treatments
  pricingModel: 'FROM_PRICE' | 'FREE' | 'PRICE_ON_CONSULTATION';
  startingPriceGBP?: number;         // e.g., 35, 60, 75, 80, 85, 95, 120
  shortSummary: string;              // Card description
  recommendedSessions?: {
    min: number;                     // e.g., 6
    max: number;                     // e.g., 8
    intervalWeeks?: number;          // e.g., 6 (for tattoo removal)
  };
  sessionDurationMinutes?: {
    min: number;                     // e.g., 10 (upper lip)
    max: number;                     // e.g., 60 (full legs/back)
  };
  requiresPatchTest: boolean;        // Default: true (24-48h prior)
}

// 3. Availability & Booking Request Entity
interface BookingRequest {
  id: string;
  treatmentId: string | 'ADVISE_ME' | 'FREE_CONSULTATION';
  preferredDate: string;             // ISO Date "YYYY-MM-DD"
  preferredTimeSlot?: string;        // Populated dynamically once date is picked
  clientDetails: {
    fullName: string;                // Required (*)
    email: string;                   // Required (*)
    phone: string;                   // Required (*)
  };
  healthScreening?: {                // Optional private fields
    medicalConditions?: string;
    currentMedications?: string;
    isPregnantOrBreastfeeding: boolean;
  };
  honeypotField?: string;            // Anti-spam ("Leave this blank")
  consents: {
    explicitDataAndHealthConsent: boolean; // Required (*) GDPR Art. 9 consent
    marketingOffersOptIn: boolean;         // Optional
  };
  status: 'PENDING_CALL' | 'CONFIRMED' | 'PATCH_TEST_DONE' | 'COMPLETED' | 'CANCELLED';
  createdAt: string;
}

// 4. Client Review / Testimonial Entity
interface Testimonial {
  id: string;
  clientName: string;                // e.g., "Sarah M."
  locationTown: string;              // e.g., "Bolton", "Bury", "Farnworth"
  treatmentName: string;             // e.g., "Laser Hair Removal"
  rating: 5;
  quote: string;
}

// 5. Local SEO Area Served Entity
interface AreaServedPage {
  id: string;
  townName: string;                  // e.g., "Bury", "Little Lever", "Bolton"
  slug: string;                      // e.g., "/areas-served/bury/"
  driveTimeSummary: string;          // e.g., "20-minute drive via the A58 or M66"
  parkingInfo: string;               // "On-street parking close to the clinic"
  featuredTreatmentIds: string[];
}

// 6. FAQ Item Entity
interface FAQItem {
  id: string;
  category: 'Getting Started' | 'Safety' | 'Cost & Courses' | 'Practical';
  question: string;
  answer: string;
}
